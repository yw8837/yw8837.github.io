인덱싱과 슬라이싱#

word = "Python"
#       0 1 2 3 4 5
#      -6-5-4-3-2-1

# 인덱싱
print(word[0])     # P
print(word[-1])    # n  (마지막 글자)
print(word[2])     # t

# 슬라이싱 [시작:끝]  ← 끝 번호는 포함 안 됨

print(beta[:3])    # abc   (처음 ~ 인덱스 2)
print(beta[4:])    # efg   (인덱스 4 ~ 끝)
print(beta[:])     # abcdefg (전체)

형 변환 (Type Casting)#

# 문자열 → 정수
print(int("42"))           # 42
print(int("3.14"))         # ValueError!  소수점 있는 문자열은 int() 직접 불가
print(int(float("3.14"))) # 3  ← float 먼저, 그 다음 int

# 정수 → 실수
print(float(7))            # 7.0

# 숫자 → 문자열
print(str(100))            # "100"
print(str(True))           # "True"

# 문자열 → 리스트
print(list("abc"))         # ['a', 'b', 'c']
print(list("파이썬"))      # ['파', '이', '썬']

섹션 3 : 리스트 메서드#

수업에서 다룬 메서드#

# 장바구니 실습
cart = ["라면", "계란", "우유"]

# append(값) — 끝에 하나 추가
cart.append("빵")
print(cart)    # ['라면', '계란', '우유', '빵']

# insert(위치, 값) — 원하는 위치에 삽입
cart.insert(1, "쌀")
print(cart)    # ['라면', '쌀', '계란', '우유', '빵']

# remove(값) — 해당 값의 첫 번째 항목 삭제
cart.remove("계란")
print(cart)    # ['라면', '쌀', '우유', '빵']

# ⚠️ 없는 값을 remove() 하면 ValueError 발생!
# cart.remove("없는항목")   → ValueError: list.remove(x): x not in list

# sort() — 정렬 (기본: 오름차순)
numbers = [5, 2, 8, 1, 9, 3]
numbers.sort()
print(numbers)                   # [1, 2, 3, 5, 8, 9]

numbers.sort(reverse=True)
print(numbers)                   # [9, 8, 5, 3, 2, 1]

⚠️ sort()는 반환값이 None입니다!
sort()는 리스트 자체를 바꾸고 아무것도 돌려주지 않습니다.
아래처럼 쓰면 result가 None이 되는 흔한 실수가 납니다.

numbers = [3, 1, 2]
result = numbers.sort()    # ❌ result = None !
print(result)              # None

# 올바른 사용법
numbers.sort()             # ✅ 리스트가 직접 정렬됨
print(numbers)             # [1, 2, 3]

# 원본을 유지하고 새 정렬 리스트가 필요하면 sorted() 사용
original = [3, 1, 2]
sorted_list = sorted(original)   # ✅ 새 리스트 반환
print(original)                  # [3, 1, 2]  (원본 유지)
print(sorted_list)               # [1, 2, 3]

시나리오 1 — 학생 성적 관리#

# 여러 학생의 점수를 리스트로 관리
students = ["홍길동", "이몽룡", "성춘향", "변학도"]
scores   = [85, 92, 78, 96]

# for 반복문 미리보기 (다음 수업 내용!)

    print(f"{students[i]}: {scores[i]}점")

시나리오 3 — 숫자 맞히기 게임#

import random

answer = random.randint(1, 100)   # 1~100 사이 랜덤 정수
count = 0

print("1~100 사이의 숫자를 맞혀보세요!")

while True:
    guess = int(input("숫자 입력: "))
    count += 1

    if guess < answer:
        print("UP! 더 큰 수입니다.")
    elif guess > answer:
        print("DOWN! 더 작은 수입니다.")
    else:
        print(f"CORRECT! 정답은 {answer}이고, {count}번 만에 맞혔습니다!")
        break

📚 공식 문서 레퍼런스#

수업에서 다루지 못했지만 알아두면 유용한 내용들입니다.
출처: Python 공식 문서 — 자료 구조

리스트 전체 메서드#

수업에서 배운 append, insert, remove, sort 외에도 다양한 메서드가 있습니다.

sample = [3, 1, 4, 1, 5, 9, 2, 6, 5, 3]

# extend(iterable) — 여러 항목을 한 번에 끝에 추가
sample.extend([7, 8])
# append([7, 8]) 는 리스트 자체가 하나로 추가되지만
# extend([7, 8]) 는 7, 8 이 각각 추가됩니다

a = [1, 2, 3]
a.append([4, 5])    # [1, 2, 3, [4, 5]]  ← 리스트가 통째로 들어감
b = [1, 2, 3]
b.extend([4, 5])    # [1, 2, 3, 4, 5]    ← 개별 항목으로 펼쳐짐

nums = [10, 20, 30, 40, 50]

# pop([인덱스]) — 해당 위치 항목을 꺼내면서 삭제, 반환값 있음
last = nums.pop()       # 인덱스 생략 시 맨 마지막 항목 꺼냄
print(last)             # 50
print(nums)             # [10, 20, 30, 40]

second = nums.pop(1)    # 인덱스 1 항목 꺼냄
print(second)           # 20
print(nums)             # [10, 30, 40]

data = [1, 2, 3, 2, 4, 2]

# count(값) — 해당 값이 몇 번 등장하는지 반환
print(data.count(2))    # 3   (2가 세 번 있음)
print(data.count(5))    # 0   (5는 없음, 에러 없이 0 반환)

# index(값) — 해당 값의 첫 번째 위치(인덱스) 반환
print(data.index(3))    # 2   (인덱스 2에 있음)
print(data.index(2))    # 1   (첫 번째 2의 위치)

⚠️ 없는 값을 index() 로 찾으면 ValueError 발생!
count()와 달리 index()는 값이 없으면 에러가 납니다.

data = [1, 2, 3]
print(data.index(99))   # ValueError: 99 is not in list

# 안전하게 쓰려면 in으로 먼저 확인
if 99 in data:
    print(data.index(99))
else:
    print("없는 값입니다.")

letters = ['d', 'a', 'c', 'b']

# reverse() — 제자리에서 순서를 뒤집음 (정렬이 아님!)
letters.reverse()
print(letters)    # ['b', 'c', 'a', 'd']

# copy() — 리스트의 얕은 복사본 반환
original = [1, 2, 3]
copied = original.copy()
copied.append(4)

print(original)   # [1, 2, 3]  ← 원본은 변하지 않음
print(copied)     # [1, 2, 3, 4]

items = [1, 2, 3, 4, 5]

# clear() — 모든 항목 삭제 (빈 리스트로 만들기)
items.clear()
print(items)      # []

리스트를 스택(Stack)으로 사용하기#

스택은 나중에 넣은 것이 먼저 나오는(LIFO: Last-In, First-Out) 자료구조입니다. 리스트의 append()와 pop()으로 간단히 구현합니다.

stack = []

stack.append("첫 번째")
stack.append("두 번째")
stack.append("세 번째")

print(stack.pop())    # 세 번째  (마지막 넣은 것이 먼저 나옴)
print(stack.pop())    # 두 번째
print(stack.pop())    # 첫 번째

리스트를 큐(Queue)로 사용하기#

큐는 먼저 넣은 것이 먼저 나오는(FIFO: First-In, First-Out) 자료구조입니다. 은행 번호표처럼 먼저 온 순서대로 처리합니다.

리스트로도 가능하지만, 앞에서 꺼내는 게 느립니다(나머지 요소를 전부 한 칸씩 이동해야 해서). 공식 문서에서는 collections.deque를 권장합니다.

from collections import deque

queue = deque(["Alice", "Bob", "Charlie"])
queue.append("Dave")      # 줄 맨 뒤에 추가
queue.append("Eve")

print(queue.popleft())    # Alice   (먼저 온 사람이 먼저 나감)
print(queue.popleft())    # Bob
print(queue)              # deque(['Charlie', 'Dave', 'Eve'])

💡 스택 vs 큐 한 눈에 비교

구분 스택(Stack) 큐(Queue)
원리 LIFO (나중에 넣은 것 먼저) FIFO (먼저 넣은 것 먼저)
비유 접시 쌓기 (맨 위가 먼저) 줄 서기 (맨 앞이 먼저)
추가 append() append()
꺼내기 pop() (뒤에서) popleft() (앞에서)

리스트 컴프리헨션 (List Comprehension)#

반복문과 조건문을 한 줄로 압축해서 리스트를 만드는 파이썬다운 문법입니다.

# 기존 방식: for 반복문으로 만들기
squares = []
for x in range(1, 6):
    squares.append(x ** 2)
print(squares)    # [1, 4, 9, 16, 25]

# 리스트 컴프리헨션으로 동일한 결과
squares = [x ** 2 for x in range(1, 6)]
print(squares)    # [1, 4, 9, 16, 25]

# 조건 추가: 짝수만 골라서 제곱
even_squares = [x ** 2 for x in range(1, 11) if x % 2 == 0]
print(even_squares)    # [4, 16, 36, 64, 100]

del 문 — 인덱스/슬라이스로 삭제#

a = [1, 2, 3, 4, 5, 6]

del a[0]        # 인덱스 0 삭제
print(a)        # [2, 3, 4, 5, 6]

del a[2:4]      # 슬라이스로 범위 삭제 (인덱스 2, 3)
print(a)        # [2, 3, 6]

💡 remove() vs del vs pop() 비교

방법 기준 반환값 예시
remove(값) 값으로 삭제 없음 a.remove(3)
del a[i] 인덱스로 삭제 없음 del a[1]
pop(i) 인덱스로 삭제 삭제된 값 a.pop(1)

집합 (Set) — 중복 없는 자료구조#

집합은 순서가 없고 중복을 허용하지 않는 자료구조입니다. 중괄호 {}로 만듭니다.

# 집합 만들기
fruits = {"사과", "바나나", "사과", "체리", "바나나"}
print(fruits)    # {'사과', '바나나', '체리'}  ← 중복 자동 제거!

# 리스트의 중복 제거에 활용
numbers = [1, 2, 2, 3, 3, 3, 4]
unique = list(set(numbers))
print(unique)    # [1, 2, 3, 4]  (순서는 보장 안 됨)

# 포함 여부 확인 (리스트보다 훨씬 빠름)
print("사과" in fruits)     # True
print("딸기" in fruits)     # False

⚠️ 주의사항

  • 집합은 순서가 없어서 인덱싱 불가: fruits[0] → TypeError
  • 빈 집합은 반드시 set()으로 생성: {} 는 빈 딕셔너리가 됩니다!
# 집합 연산 (수학의 집합과 동일)
a = {1, 2, 3, 4}
b = {3, 4, 5, 6}

print(a | b)    # {1, 2, 3, 4, 5, 6}  합집합 (둘 중 하나라도 있으면)
print(a & b)    # {3, 4}              교집합 (둘 다 있는 것)
print(a - b)    # {1, 2}              차집합 (a에만 있는 것)

튜플 (Tuple) — 변경 불가 리스트#

리스트와 비슷하지만 한 번 만들면 수정할 수 없는 자료구조입니다. 소괄호 ()를 사용합니다.

# 튜플 생성
point = (10, 20)
colors = ("red", "green", "blue")

# 인덱싱, 슬라이싱은 리스트와 동일
print(colors[0])     # red
print(colors[1:])    # ('green', 'blue')

# 수정 불가
colors[0] = "yellow"  # TypeError: 'tuple' object does not support item assignment

# 패킹(packing): 여러 값을 하나의 튜플로
packed = 1, 2, 3
print(packed)    # (1, 2, 3)

# 언패킹(unpacking): 튜플을 여러 변수에 분배
x, y, z = packed
print(x, y, z)   # 1 2 3

딕셔너리 (Dictionary) — 키-값 쌍#

중괄호 {}로 만들며, 키(key)로 값(value)에 접근합니다.

# 딕셔너리 생성
person = {
    "name": "홍길동",
    "age": 25,
    "skills": ["Python", "Java"]
}

# 값 접근
print(person["name"])      # 홍길동
print(person["skills"])    # ['Python', 'Java']

# 값 추가 / 수정
person["email"] = "hong@example.com"    # 새 키 추가
person["age"] = 26                       # 기존 값 수정

# 키 존재 여부 확인
print("name" in person)     # True
print("phone" in person)    # False

# 반복
for key, value in person.items():
    print(f"{key}: {value}")
