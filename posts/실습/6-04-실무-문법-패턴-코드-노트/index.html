실무에서 자주 쓰는 심화 문법 6가지를 코드로 익힌 실습 노트입니다. 강의 노트북 코드를 그대로 정리하고 한 줄씩 풀었어요. 🧩

1. f-string 포매팅#

name, role, salary = 'Alice', '시니어 개발자', 6500
print(f"이름: {name}, 직책: {role}, 연봉: {salary}만원")

# 식·조건도 가능
a, b = 15, 4
print(f"{a} + {b} = {a + b}")          # 15 + 4 = 19
score = 85
print(f"점수: {score}점 -> {'합격' if score >= 60 else '불합격'}")

# 숫자 포맷
pi, big, rate = 3.141592, 1234567890, 0.12345
print(f"π = {pi:.2f}")          # 3.14
print(f"큰 숫자: {big:,}")       # 1,234,567,890
print(f"달성률: {rate:.1%}")     # 12.3%
print(f"연봉: {65000000:,}원")   # 65,000,000원

# 정렬 (로그 출력)
import datetime
def log(level, msg):
    now = datetime.datetime.now().strftime('%H:%M:%S')
    print(f'[{now}] {level:>7} | {msg}')      # 레벨 오른쪽 7칸 정렬
log('INFO', '서버 시작')
log('ERROR', 'DB 연결 실패')

출력:

[14:12:51]    INFO | 서버 시작
[14:12:51]   ERROR | DB 연결 실패

2. 리스트 컴프리헨션#

# for문 → 컴프리헨션
squares = [i ** 2 for i in range(1, 11)]
print(squares)   # [1, 4, 9, 16, 25, 36, 49, 64, 81, 100]

members = [
    {'name': 'Alice', 'team': '개발팀', 'salary': 6500},
    {'name': 'Bob',   'team': '기획팀', 'salary': 5200},
    {'name': 'Derek', 'team': '개발팀', 'salary': 4800},
    {'name': 'Eve',   'team': '분석팀', 'salary': 3600},
]

# 필터: 연봉 5000+ 이름
high = [m['name'] for m in members if m['salary'] >= 5000]
print(high)                       # ['Alice', 'Bob']

# 개발팀만 + f-string
dev = [f"{m['name']}({m['salary']:,}만원)" for m in members if m['team'] == '개발팀']
print(dev)                        # ['Alice(6,500만원)', 'Derek(4,800만원)']

# if-else 변환 (순서 주의!)
grade = [f"{m['name']}: 고연봉" if m['salary'] >= 5000 else f"{m['name']}: 일반" for m in members]

# 실무: 데이터 전처리 (공백 제거 + 빈 값 필터)
raw = [' Python ', 'Django ', ' React', 'SQL', ' ', '']
clean = [s.strip() for s in raw if s.strip()]
print(clean)      # ['Python', 'Django', 'React', 'SQL']

# 딕셔너리 컴프리헨션
salary_dict = {m['name']: m['salary'] for m in members}
print(salary_dict)   # {'Alice': 6500, 'Bob': 5200, 'Derek': 4800, 'Eve': 3600}

⚠️ 필터 if는 뒤(for x in seq if 조건), if-else 변환은 앞(A if 조건 else B for x in seq).

3. sorted — 정렬#

scores = [85, 92, 78, 96, 73, 88]
print(sorted(scores))                 # [73, 78, 85, 88, 92, 96]  원본 유지
print(sorted(scores, reverse=True))   # [96, 92, 88, 85, 78, 73]

scores.sort()      # 원본 자체를 변경, 반환은 None

members = [
    {'name': 'Alice', 'team': '개발팀', 'salary': 6500},
    {'name': 'Carol', 'team': '영업팀', 'salary': 6100},
    {'name': 'Eve',   'team': '분석팀', 'salary': 3600},
    {'name': 'Helen', 'team': '분석팀', 'salary': 5500},
]

# key=lambda 로 정렬 기준 지정
by_salary = sorted(members, key=lambda m: m['salary'], reverse=True)
for m in by_salary:
    print(f" {m['name']:<8} {m['salary']:,}만원")

# 다중 기준: 팀명 오름차순 → 같은 팀 연봉 내림차순
multi = sorted(members, key=lambda m: (m['team'], -m['salary']))

# 글자 수 기준
words = ['Python', 'AI', 'MachineLearning', 'SQL']
print(sorted(words, key=len))   # ['AI', 'SQL', 'Python', 'MachineLearning']

4. enumerate — 인덱스 + 값#

for idx, season in enumerate(['봄', '여름', '가을', '겨울']):
    print(f" {idx}: {season}")

# start=1 (1번부터)
for idx, team in enumerate(['개발팀', '기획팀'], start=1):
    print(f" {idx}번 팀: {team}")

# 실무: 정렬 후 순위 매기기
scores = [{'name':'Alice','score':95}, {'name':'Bob','score':82}, {'name':'Derek','score':91}]
ranked = sorted(scores, key=lambda m: m['score'], reverse=True)

medals = {1:'🥇', 2:'🥈', 3:'🥉'}
for rank, m in enumerate(ranked, 1):
    print(f"  {medals.get(rank, str(rank)+'위')} {m['name']:<8} {m['score']}점")

출력:

  🥇 Alice    95점
  🥈 Derek    91점
  🥉 Bob      82점

5. lambda · map · filter#

square = lambda x: x ** 2        # 이름 없는 한 줄 함수
print(square(5))                 # 25

nums = [1, 2, 3, 4, 5]
print(list(map(lambda x: x*2, nums)))         # [2, 4, 6, 8, 10]
print(list(filter(lambda x: x%2==0, nums)))   # [2, 4]  (짝수만)

# 실무: 연봉 원 단위 → 만원 단위
salaries = [65000000, 52000000, 48000000]
man = list(map(lambda w: w // 10000, salaries))
print(man)     # [6500, 5200, 4800]

# 복잡한 로직은 일반 함수로 (lambda 한계)
def calculate_bonus(m):
    base = m['salary'] * 0.1
    if m.get('years', 1) >= 5: base *= 1.05
    if m.get('score', 80) >= 90: base *= 1.1
    return int(base)

💡 map/filter는 list()로 감싸야 결과가 보입니다. 간단하면 lambda, 복잡하면 def.

6. if __name__ == "__main__":#

def greet(name):
    return f'안녕하세요, {name}님!'

if __name__ == '__main__':       # 직접 실행할 때만 동작
    print(greet('Alice'))        # import로 불러오면 실행 안 됨
    print('메인 실행 완료!')

💡 함수·클래스는 어디서 import해도 쓰되, “직접 실행 코드"는 이 블록 안에 둬서 분리합니다.

📚 보충#

Python 공식 문서 기반.

  • map vs 컴프리헨션: 같은 일 → 보통 [x*2 for x in nums](컴프리헨션)이 더 읽기 쉬워 선호.
  • sys.exit(main()) 패턴: .py 진입점에서 main() 결과를 종료코드로 반환(0=정상). 주피터에선 커널이 죽으니 주의.
  • 정렬 안정성: sorted는 같은 key값의 원래 순서를 보존(stable). 다중 정렬에 활용.

핵심 정리#

문법 형식
f-string f"{값:.2f}" {값:,} {값:>7}
컴프리헨션 [식 for x in seq if 조건]
sorted sorted(list, key=lambda, reverse=True)
enumerate for i, v in enumerate(list, start=1)
lambda/map list(map(lambda x: ..., list))
__name__ if __name__ == "__main__":
